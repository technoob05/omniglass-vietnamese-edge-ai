\section{Động lực}

\begin{frame}{Vì sao chọn on-device, không phải cloud?}
  \begin{block}{Bài toán}
    Một trợ lý cảnh báo nguy hiểm cho người khiếm thị không thể chấp nhận độ trễ
    mạng, mất kết nối, hay để hình ảnh camera rời khỏi thiết bị.
  \end{block}
  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\faCloud\ Trợ lý VLM trên cloud}
      \begin{itemize}
        \item Độ trễ round-trip: hàng trăm ms tới vài giây
        \item Cần kết nối mạng liên tục
        \item Dữ liệu camera/mic phải rời khỏi thiết bị
        \item Chi phí tăng theo lượng sử dụng
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\faMicrochip\ OmniGlass Edge}
      \begin{itemize}
        \item Detection + depth chạy \textbf{cục bộ}, từng khung hình
        \item Hoạt động hoàn toàn offline
        \item Camera/mic không rời khỏi box
        \item Lập lịch có ý thức về nhiệt độ, an toàn là ưu tiên số một
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Nhận dạng giọng nói (STT)}

\begin{frame}{STT: Ba hướng đã thử, một hướng lên bản chính}
  \begin{enumerate}[<+->]
    \item \textbf{GPU (Adreno 740, OpenCL)} -- thiết bị nhận diện đúng, nhưng
          \emph{cả hai} kernel OpenCL Q8 và FP16 của whisper.cpp đều segfault
          (\texttt{exit 139}). Loại bỏ: sẽ làm STT crash khi lên sản phẩm.
    \item \textbf{NPU (Hexagon HTP, QNN)} -- hướng đúng về lâu dài; Whisper cần một
          QNN context được export qua Qualcomm AI Hub. Box hiện chưa có token/credential
          AI Hub $\Rightarrow$ bị chặn, chưa bị bỏ.
    \item \textbf{Streaming thô} (\texttt{whisper-stream}, chunk 1,5\,s) -- tự sinh câu
          tiếng Việt giả khi phòng im lặng, mỗi chunk mã hóa mất $\approx$2,7\,s.
          Không an toàn cho thiết bị hỗ trợ người khiếm thị. Loại bỏ.
  \end{enumerate}
  \begin{alertblock}<4->{Cấu hình đã lên bản chính}
    \texttt{whisper.cpp} trên CPU/NEON, \textbf{Base Q8} (nâng từ Small Q8),
    Silero VAD (ngưỡng 0,5, tối thiểu nói 250\,ms, tối thiểu im lặng 400\,ms),
    audio context chốt 512 frame, chỉ trả transcript \emph{cuối cùng} sau khi
    thả nút (VAD-bounded final).
  \end{alertblock}
\end{frame}

\begin{frame}{STT: Kết quả đo thực tế}
  \centering
  \begin{tabular}{lccc}
    \toprule
    \textbf{Cấu hình} & \textbf{Audio} & \textbf{Thời gian xử lý} & \textbf{RTF} \\
    \midrule
    Small Q8 (bản cũ) & 8,75\,s & độ trễ cao hơn & -- \\
    \textbf{Base Q8 + VAD ctx-512} & 8,75\,s & \textbf{2,48\,s} & \textbf{0,28} \\
    \bottomrule
  \end{tabular}
  \vspace{10pt}

  RTF~0,28 nghĩa là một đoạn ghi âm mic thật dài 8,75 giây được nhận dạng
  chỉ trong khoảng 2,5 giây -- nhanh hơn thời gian thực rõ rệt, chỉ dùng CPU,
  trong khi YOLO/depth vẫn chạy nguyên vẹn trên NPU.
\end{frame}

\section{Suy luận ngôn ngữ (VLM)}

\begin{frame}{VLM: Qwen3.5 2B VL}
  \begin{itemize}
    \item Backend: \texttt{llama\_cpp} (nhãn trên pipeline: \emph{GenieX Qwen3.5 2B VL}).
    \item Chỉ được gọi \emph{theo yêu cầu} -- một lượt cho mỗi lần thả nút
          push-to-talk, không bao giờ chạy trong vòng lặp always-on.
    \item Prompt luôn được ``bám thực tế'' bằng scene facts sống từ detector/depth
          (loại vật thể, khoảng cách, nguy hiểm), không chỉ gửi khung hình thô.
    \item Việc được phép chạy (admission) bị chặn bởi cơ chế nhiệt NPU chia sẻ
          (xem phần sau).
  \end{itemize}
\end{frame}

% =============================================================================
\section{Tổng hợp giọng nói (TTS)}

\begin{frame}{TTS: VieNeu, phát streaming}
  \begin{itemize}
    \item Backend \texttt{vieneu}, giọng \textbf{Phạm Tuyên}, tốc độ phát
          \textbf{1,5$\times$}.
    \item \textbf{Streaming, không phải batch:} PCM theo từng cụm câu
          (\texttt{phrase\_pcm}) ở 48\,kHz -- cụm âm đầu tiên bắt đầu phát trước
          khi toàn bộ câu trả lời tổng hợp xong.
    \item Hai đích phát độc lập: loa ALSA trên box, hoặc Web Audio API của
          trình duyệt để theo dõi từ xa khi phát triển.
    \item Cache kết quả quan sát trực tiếp: \texttt{cache\_hits=4,
          cache\_misses=0} với các cụm câu lặp lại.
  \end{itemize}
\end{frame}

\section{Lập lịch an toàn là ưu tiên số một}

\begin{frame}[fragile]{Cơ chế chặn nhiệt (Thermal Admission Guard)}
  \begin{block}{Quy tắc}
    Lượt VLM chỉ được \textbf{admit} khi \texttt{npu\_temperature\_c}
    $\le 75.0^\circ$C. Perception, STT, TTS \textbf{không} bị ràng buộc này.
  \end{block}
  \pause
  \vspace{6pt}
  \textbf{Quan sát thực tế (hai lần kiểm tra độc lập, cùng một buổi):}
  \begin{lstlisting}
"npu_temperature_c": 79.2  -> vlm_admitted: false
"npu_temperature_c": 80.8  -> vlm_admitted: false
"vlm_admission_reasons": ["npu_temperature_c=80.8 above 75.0"]
\end{lstlisting}
  \pause
  \vspace{6pt}
  \textbf{Vì sao quan trọng với thiết bị hỗ trợ người khiếm thị:} một lượt VLM
  ``nói nhiều'' không bao giờ được phép làm nghẽn hay crash vòng lặp phát hiện
  nguy hiểm. Khi box nóng, OmniGlass Edge \emph{giảm cấp một cách an toàn} --
  câu hỏi xếp hàng chờ, còn detection vẫn chạy liên tục.
\end{frame}

\section{Hạn chế \& hướng phát triển}

\begin{frame}{Hạn chế \& hướng phát triển}
  \begin{itemize}
    \item \textbf{Streaming ASR thật} cần một QNN context của Whisper export qua
          Qualcomm AI Hub (công bố của hãng: encoder Whisper-Tiny $\approx$18\,ms
          trên NPU của QCS8550) -- đang chặn ở credential AI Hub, không phải ở
          kiến trúc.
    \item Backend GPU (Adreno 740) OpenCL cho whisper.cpp vẫn còn segfault --
          chưa cô lập được root cause.
    \item Push-to-talk chỉ single-turn: chưa có barge-in / ngắt lời khi box
          đang nói.
    \item Chưa có đánh giá WER (word error rate) chính thức -- kết quả hiện tại
          chỉ là độ trễ/throughput.
    \item Hiện chỉ tiếng Việt; đa ngôn ngữ là scope creep chưa nằm trong kế
          hoạch.
  \end{itemize}
\end{frame}

\section{Kết luận}

\begin{frame}{Điểm chính cần nhớ}
  \begin{itemize}
    \item \textbf{OmniGlass Edge} chạy toàn bộ vòng lặp trợ giúp --
          phát hiện $\to$ hiểu $\to$ nói -- hoàn toàn trên một box Qualcomm
          QCS8550, không cần round-trip qua cloud.
    \item Mọi con số trong bộ slide này đều được xác thực bằng một lần gọi
          \texttt{/health} thật hoặc một lượt test pass thật, không phải từ
          spec sheet.
    \item An toàn được đưa vào \emph{kiến trúc}: perception luôn giữ ưu tiên,
          VLM bị chặn theo nhiệt, và một hướng streaming STT nhanh-nhưng-sai đã
          bị loại bỏ có chủ đích.
  \end{itemize}
